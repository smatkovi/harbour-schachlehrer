# Sailfish OS build of harbour-schachlehrer (Qt 5.6, Silica).
#
# docs/design.md §2 is the map of the tree: src/core/ is the Qt-free, testable
# core, src/ is the Qt layer with the single QML facade TeacherEngine, and
# third_party/chess-library/ holds the vendored Disservin/chess-library header.
#
# The C++ arrives milestone by milestone (design.md §8, M0..M8). Neither a
# source file that is not written yet nor one that is added later may break the
# configure step of everything else, so the sources are globbed and every test
# is guarded by if(EXISTS). After adding a file in a working tree, re-run cmake;
# the RPM and every clean build evaluate the glob anyway.
cmake_minimum_required(VERSION 3.6)
project(harbour-schachlehrer LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

find_package(Qt5 REQUIRED COMPONENTS Core LinguistTools Network Qml Quick Sql)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SAILFISHAPP REQUIRED sailfishapp)

set(SCHACH_APP harbour-schachlehrer)
set(SCHACH_DATADIR share/${SCHACH_APP})

# --------------------------------------------------------------------------
# Sources
# --------------------------------------------------------------------------

# The Qt-free core of docs/design.md §3: Position, Uci, Taxonomy, WinProb,
# Skill, Placement, Srs, Card. No Qt, no I/O, no clock — that is what makes the
# tests under tests/ possible with fixed inputs.
file(GLOB SCHACH_CORE_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/core/*.cpp)
list(SORT SCHACH_CORE_SOURCES)

# The Qt layer of §2: EngineProcess, Database, Analyser, Sparring and the one
# QML facade TeacherEngine. main.cpp is pulled out again below so the tests can
# link the same objects without a second entry point.
file(GLOB SCHACH_QT_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
list(SORT SCHACH_QT_SOURCES)
set(SCHACH_MAIN ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp)
list(REMOVE_ITEM SCHACH_QT_SOURCES ${SCHACH_MAIN})

if(EXISTS ${SCHACH_MAIN})
    set(SCHACH_ENTRY ${SCHACH_MAIN})
else()
    # M0 is still being written next door. Keep the configure step green and
    # say so instead of failing on a file that is on its way.
    # add_executable() needs at least one translation unit. A stub keeps the
    # configure step green so the packaging, the QML and the installs can be
    # checked before M0 has landed; the link then fails on the missing main(),
    # which is the honest error.
    message(STATUS "${SCHACH_APP}: src/main.cpp is not there yet — stubbing the entry point")
    set(SCHACH_ENTRY ${CMAKE_CURRENT_BINARY_DIR}/schach_missing_main.cpp)
    file(WRITE ${SCHACH_ENTRY}
         "// Generated by CMakeLists.txt because src/main.cpp does not exist yet.\n"
         "// Delete the build directory once it does.\n")
endif()

add_executable(${SCHACH_APP}
    ${SCHACH_ENTRY}
    ${SCHACH_CORE_SOURCES}
    ${SCHACH_QT_SOURCES}
)

target_include_directories(${SCHACH_APP} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/chess-library
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/chess-library/include
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/chess-library/src
    ${SAILFISHAPP_INCLUDE_DIRS}
)

target_link_libraries(${SCHACH_APP}
    Qt5::Core Qt5::Network Qt5::Qml Qt5::Quick Qt5::Sql ${SAILFISHAPP_LIBRARIES}
)

# --------------------------------------------------------------------------
# Translations
# --------------------------------------------------------------------------
# The interface language is German and the source strings are German too
# (docs/design.md §8, M7). The catalogue exists so that every user-visible
# string stays extractable and a second language can be added without touching
# the QML again.
file(GLOB SCHACH_TRANSLATIONS ${CMAKE_CURRENT_SOURCE_DIR}/translations/*.ts)
list(SORT SCHACH_TRANSLATIONS)
if(SCHACH_TRANSLATIONS)
    qt5_add_translation(SCHACH_QM_FILES ${SCHACH_TRANSLATIONS})
    add_custom_target(${SCHACH_APP}-translations ALL DEPENDS ${SCHACH_QM_FILES})
    add_dependencies(${SCHACH_APP} ${SCHACH_APP}-translations)
    install(FILES ${SCHACH_QM_FILES} DESTINATION ${SCHACH_DATADIR}/translations)
endif()

# --------------------------------------------------------------------------
# The engine (platform.md §1.9, §6.1)
# --------------------------------------------------------------------------
# Stockfish is not built here. It is cross-built out of tree (platform.md §8.3)
# and dropped into third_party/prebuilt/<arch>/stockfish; the spec passes the
# RPM architecture in. It is installed as a plain data file with the executable
# bit — a separate UCI process started with QProcess, not a linked library,
# which is what keeps the GPL question of platform.md §1.7 simple and what lets
# the UI survive an engine crash.
set(SCHACH_ENGINE_ARCH "${CMAKE_SYSTEM_PROCESSOR}" CACHE STRING
    "Architecture directory under third_party/prebuilt that holds stockfish")
option(SCHACH_REQUIRE_ENGINE
    "Fail the configure step when the prebuilt engine for SCHACH_ENGINE_ARCH is missing" OFF)

set(SCHACH_ENGINE
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/prebuilt/${SCHACH_ENGINE_ARCH}/stockfish)

if(EXISTS ${SCHACH_ENGINE})
    install(PROGRAMS ${SCHACH_ENGINE} DESTINATION ${SCHACH_DATADIR}/bin)
    # The same binary under /usr/bin. chess-spec/platform.md §7.1 names an exec
    # from /usr/share as the one thing that could sink the process
    # architecture under Sailjail; this is the documented way out, and the app
    # prefers it. It also settles rpmlint's arch-dependent-file complaint.
    install(PROGRAMS ${SCHACH_ENGINE} DESTINATION ${CMAKE_INSTALL_PREFIX}/bin
            RENAME ${SCHACH_APP}-engine)
    message(STATUS "${SCHACH_APP}: engine ${SCHACH_ENGINE}")
elseif(SCHACH_REQUIRE_ENGINE)
    message(FATAL_ERROR
        "${SCHACH_APP}: no engine at third_party/prebuilt/${SCHACH_ENGINE_ARCH}/stockfish.\n"
        "Cross-build Stockfish 17.1 for ${SCHACH_ENGINE_ARCH} as described in\n"
        "chess-spec/platform.md §8.3 and copy the stripped binary there.")
else()
    message(WARNING
        "${SCHACH_APP}: no engine at third_party/prebuilt/${SCHACH_ENGINE_ARCH}/stockfish — "
        "building without it. The app treats a missing engine as a normal state "
        "(design.md §5); the RPM does not.")
endif()

# The small NNUE net that goes with the -DNNUE_EMBEDDING_OFF build
# (platform.md §1.4): 3,36 MiB instead of the 71 MiB big net, and 57 MiB RSS
# instead of 273. tools/fetch-engine.sh leaves it in assets/; a net built into
# the engine would make it unnecessary, hence the glob rather than a hard name.
file(GLOB SCHACH_NET
    ${CMAKE_CURRENT_SOURCE_DIR}/assets/*.nnue
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/prebuilt/net/*.nnue)
if(SCHACH_NET)
    install(FILES ${SCHACH_NET} DESTINATION ${SCHACH_DATADIR}/net)
    message(STATUS "${SCHACH_APP}: NNUE net ${SCHACH_NET}")
else()
    message(WARNING
        "${SCHACH_APP}: no *.nnue in assets/ — an engine built with "
        "-DNNUE_EMBEDDING_OFF cannot evaluate without it (platform.md §1.4)")
endif()

# --------------------------------------------------------------------------
# Endgame tablebases (platform.md §2, Empfehlung 2)
# --------------------------------------------------------------------------
# Syzygy 3+4 men, WDL and DTZ, 70 files, 4 346 080 B. Handed to the engine as
# `setoption name SyzygyPath value /usr/share/harbour-schachlehrer/syzygy`.
file(GLOB SCHACH_SYZYGY
    ${CMAKE_CURRENT_SOURCE_DIR}/assets/syzygy/*.rtbw
    ${CMAKE_CURRENT_SOURCE_DIR}/assets/syzygy/*.rtbz)
list(SORT SCHACH_SYZYGY)
list(LENGTH SCHACH_SYZYGY SCHACH_SYZYGY_COUNT)
if(SCHACH_SYZYGY)
    install(FILES ${SCHACH_SYZYGY} DESTINATION ${SCHACH_DATADIR}/syzygy)
    message(STATUS "${SCHACH_APP}: ${SCHACH_SYZYGY_COUNT} syzygy files (expected 70)")
else()
    message(WARNING "${SCHACH_APP}: assets/syzygy/ is empty — no tablebases will be packaged")
endif()

# --------------------------------------------------------------------------
# QML, graphics, desktop entry, icons
# --------------------------------------------------------------------------
# Silica pages and qmldir land in one directory, the way `import "."` expects
# them (docs/design.md §7).
file(GLOB SCHACH_QML ${CMAKE_CURRENT_SOURCE_DIR}/sailfish/*.qml)
list(SORT SCHACH_QML)
install(FILES ${SCHACH_QML} ${CMAKE_CURRENT_SOURCE_DIR}/sailfish/qmldir
        DESTINATION ${SCHACH_DATADIR}/qml)
# AboutPage.qml shows the icon; one copy next to the QML is enough, the other
# four sizes live under hicolor only (rpmlint files-duplicate).
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/sailfish/icons/icon-256.png
        DESTINATION ${SCHACH_DATADIR}/qml/icons)

# cburnett piece SVGs, BSD-3-Clause, see CREDITS/ASSETS.md (platform.md §4.3).
install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/assets/pieces
        DESTINATION ${SCHACH_DATADIR}/assets FILES_MATCHING PATTERN "*.svg")

# The placement items (teacher.md §4.2). Without them the test says so rather
# than asking about the starting position, so a missing bank is a warning.
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/assets/items/placement.json)
    install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/assets/items
            DESTINATION ${SCHACH_DATADIR} FILES_MATCHING PATTERN "*.json")
else()
    message(WARNING "${SCHACH_APP}: assets/items/placement.json is missing — "
                    "the placement test will have nothing to ask")
endif()

install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/sailfish/desktop/${SCHACH_APP}.desktop
        DESTINATION share/applications)

foreach(size 86 108 128 172 256)
    install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/sailfish/icons/icon-${size}.png
            DESTINATION share/icons/hicolor/${size}x${size}/apps
            RENAME ${SCHACH_APP}.png)
endforeach()

install(TARGETS ${SCHACH_APP} RUNTIME DESTINATION bin)

foreach(doc CREDITS/ASSETS.md README.md)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${doc})
        install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/${doc} DESTINATION share/doc/${SCHACH_APP})
    endif()
endforeach()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/LICENSE)
    install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/LICENSE DESTINATION share/licenses/${SCHACH_APP})
endif()

# --------------------------------------------------------------------------
# Tests (docs/design.md §2, §8)
# --------------------------------------------------------------------------
# perft, the UCI parser, the taxonomy, FSRS and the placement test. Each is its
# own target over the Qt-free core only — that is the whole point of keeping
# src/core/ free of Qt. A test whose source has not been written yet is
# reported and skipped, never a configure error.
option(SCHACH_BUILD_TESTS "Build the core, perft, taxonomy, FSRS and placement tests" OFF)
if(SCHACH_BUILD_TESTS)
    enable_testing()

    set(SCHACH_TESTS
        test_perft test_perft_epd test_position test_san test_pgn
        test_uci_parser test_taxonomy test_winprob test_skill
        test_placement test_srs test_card)
    # These two need the Qt layer, not only the Qt-free core: the fair-play
    # lock of platform.md §3.7 lives in EngineProcess and TeacherEngine, and
    # the Lichess protocol lives in Lichess/GameSync. They link Network and Sql
    # as well and get SCHACH_SOURCE_DIR so test_fairplay can grep the tree for
    # the Bot API upgrade endpoint that must not exist (§3.1).
    set(SCHACH_QT_TESTS test_fairplay test_lichess test_session)
    # Anything else the other author drops into tests/ is picked up as well.
    file(GLOB SCHACH_TEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.cpp)
    foreach(src IN LISTS SCHACH_TEST_SOURCES)
        get_filename_component(name ${src} NAME_WE)
        list(APPEND SCHACH_TESTS ${name})
    endforeach()
    list(REMOVE_DUPLICATES SCHACH_TESTS)
    list(SORT SCHACH_TESTS)

    foreach(t IN LISTS SCHACH_TESTS)
        if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/${t}.cpp)
            list(FIND SCHACH_QT_TESTS ${t} schach_qt_test)
            if(schach_qt_test GREATER -1)
                add_executable(${t} tests/${t}.cpp ${SCHACH_CORE_SOURCES} ${SCHACH_QT_SOURCES})
            else()
                add_executable(${t} tests/${t}.cpp ${SCHACH_CORE_SOURCES})
            endif()
            target_include_directories(${t} PRIVATE
                ${CMAKE_CURRENT_SOURCE_DIR}/src
                ${CMAKE_CURRENT_SOURCE_DIR}/third_party/chess-library
                ${CMAKE_CURRENT_SOURCE_DIR}/third_party/chess-library/include
                ${CMAKE_CURRENT_SOURCE_DIR}/third_party/chess-library/src)
            target_compile_definitions(${t} PRIVATE
                SCHACH_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
                SCHACH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
            if(schach_qt_test GREATER -1)
                target_link_libraries(${t} Qt5::Core Qt5::Network Qt5::Sql)
            else()
                target_link_libraries(${t} Qt5::Core)
            endif()
            add_test(NAME ${t} COMMAND ${t})
            # test_session finds the item bank through the environment.
            set_tests_properties(${t} PROPERTIES
                ENVIRONMENT "SCHACH_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}")
        else()
            message(STATUS "${SCHACH_APP}: tests/${t}.cpp is not there yet, skipping it")
        endif()
    endforeach()
endif()
